from flask import Flask, render_template, jsonify, request from bitcoin_address_generator import generate_wallet, generate_wallets, validate_bitcoin_address app = Flask(__name__) @app.route("/", methods=["GET", "POST"]) def index(): count = request.form.get("count", 1, type=int) if count < 1: count = 1 wallets = generate_wallets(count) return render_template("index.html", wallets=wallets, count=count) @app.route("/api/generate") def api_generate(): count = request.args.get("count", default=1, type=int) if count < 1: count = 1 return jsonify({"wallets": generate_wallets(count)}) @app.route("/api/validate") def api_validate(): address = request.args.get("address", "") return jsonify({"valid": validate_bitcoin_address(address), "address": address}) if __name__ == "__main__": app.run(debug=True)
